    <script>
        // Функции для чата (заглушки для первого шага)
        function sendMessage() {
            const userInput = document.querySelector('.user-window').value;
            if (userInput) {
                document.getElementById('botResponse').value = `Бот генерирует пост на основе: "${userInput.substring(0, 50)}..."`;
                // Здесь будет логика бота в следующих шагах
            }
        }

        function acceptPost() {
            alert('Пост принят!');
            // Логика принятия
        }

        function sendFeedback() {
            const feedback = document.getElementById('feedback').value;
            if (feedback) {
                document.getElementById('botResponse').value += `\n\nУчтены замечания: "${feedback.substring(0, 50)}..."`;
                document.getElementById('feedback').value = '';
            }
        }

        function acceptFeedback() {
            alert('Замечания приняты!');
        }

        // Интерактивный календарь
        const months = ['Январь', 'Февраль', 'Март', 'Апрель', 'Май', 'Июнь', 'Июль', 'Август', 'Сентябрь', 'Октябрь', 'Ноябрь', 'Декабрь'];
        const daysOfWeek = ['Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб', 'Вс'];

        let currentDate = new Date(2026, 4, 14); // Май 2026
        let currentMonth = currentDate.getMonth();
        let currentYear = currentDate.getFullYear();

        function createCalendar(elementId, monthOffset) {
            const calendar = document.getElementById(elementId);
            const month = (currentMonth + monthOffset + 12) % 12;
            const year = currentYear + Math.floor((currentMonth + monthOffset) / 12);
            const today = new Date();

            calendar.innerHTML = `
                <div class="calendar-header">
                    <button class="calendar-nav" onclick="prevMonth(${monthOffset})">‹</button>
                    <h3>${months[month]} ${year}</h3>
                    <button class="calendar-nav" onclick="nextMonth(${monthOffset})">›</button>
                </div>
                <div class="calendar-grid">
                    ${daysOfWeek.map(day => `<div class="calendar-header-row">${day}</div>`).join('')}
                </div>
                <div class="calendar-grid" id="dates-${elementId}"></div>
            `;

            const firstDay = new Date(year, month, 1).getDay() || 7; // Пн=1
            const daysInMonth = new Date(year, month + 1, 0).getDate();
            const datesGrid = document.getElementById(`dates-${elementId}`);

            // Пустые дни
            for (let i = 1; i < firstDay; i++) {
                const day = document.createElement('div');
                day.className = 'calendar-day';
                datesGrid.appendChild(day);
            }

            // Дни месяца
            for (let day = 1; day <= daysInMonth; day++) {
                const dateEl = document.createElement('div');
                dateEl.className = 'calendar-day';
                dateEl.textContent = day;
                if (year === today.getFullYear() && month === today.getMonth() && day === today.getDate()) {
                    dateEl.classList.add('today');
                }
                dateEl.onclick = () => alert(`Выбрана дата: ${day} ${months[month]} ${year}`);
                datesGrid.appendChild(dateEl);
            }
        }

        function renderCalendars() {
            createCalendar('currentMonth', 0);
            createCalendar('nextMonth', 1);
        }

        function prevMonth(offset) {
            currentMonth = (currentMonth - 1 + 12) % 12;
            if (offset === 0) currentYear = currentMonth < 4 ? currentYear - 1 : currentYear; // Примерная логика
            renderCalendars();
        }

        function nextMonth(offset) {
            currentMonth = (currentMonth + 1) % 12;
            if (offset === 0 && currentMonth === 0) currentYear++;
            renderCalendars();
        }

        // Инициализация
        renderCalendars();
    </script>
